type acceptanceValueKey struct{}
type acceptanceCredentials struct { fail error; wait bool; sawValue atomic.Bool }
func (c *acceptanceCredentials) GetAuth(ctx context.Context, name string) (string,error) {
 if name != cancelTaskAuthToolName { return "",nil }
 c.sawValue.Store(ctx.Value(acceptanceValueKey{}) == "preserved")
 if c.wait { <-ctx.Done() }
 return "",c.fail
}
func TestAcceptanceAConcurrentCallbackParent(t *testing.T) {
 // Arrange: one shared client, independent host invocations and atomic observer state.
 var creates,cancels,observed atomic.Int32
 parentCause:=errors.New("host custom cause")
 callbackErr:=errors.New("consumer exact error")
 s:=httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter,r *http.Request){
  _,_=io.Copy(io.Discard,r.Body)
  switch {
  case strings.HasSuffix(r.URL.Path,"/cancel"): cancels.Add(1); w.WriteHeader(204)
  case strings.HasSuffix(r.URL.Path,"/tasks"): creates.Add(1); w.WriteHeader(201); _,_=w.Write([]byte(`{"task_id":"t","artifacts":[]}`))
  default: w.Header().Set("Content-Type","text/event-stream"); _,_=w.Write([]byte(stepFrame("s","running",false))); w.(http.Flusher).Flush(); <-r.Context().Done()
  }
 }))
 defer s.Close()
 c:=mustClient(t,s.URL,WithAllowPrivateIPs(true),WithCancellationObserver(func(ctx context.Context,d CancellationDiagnostic){
  observed.Add(1)
  if d.ParentInterrupt!=context.Canceled || d.ParentCause!=parentCause || d.Cause!=nil || !d.Acknowledged || d.Stage!=CancellationRequest { t.Errorf("bad diagnostic: %+v",d) }
  if ctx.Value(acceptanceValueKey{})!="preserved" || ctx.Err()!=nil { t.Error("cleanup values or lifetime lost") }
 }))
 tool,err:=AsTool("delegate","remote",[]byte(`{"type":"object"}`),c); if err!=nil { t.Fatal(err) }
 const calls=12
 var wg sync.WaitGroup
 // Act: every callback cancels its own parent, then returns the exact consumer error.
 for range calls { wg.Go(func(){
  ctx,cancel:=context.WithCancelCause(context.WithValue(t.Context(),acceptanceValueKey{},"preserved")); defer cancel(nil)
  err:=tool.Execute(ctx,toolsy.NewRunEnv(nil),toolsy.ToolInput{ArgsJSON:[]byte(`{}`)},func(ch toolsy.Chunk)error{
   if ch.Event==toolsy.EventResult { t.Error("unexpected model result") }
   cancel(parentCause); return callbackErr
  })
  if !errors.Is(err,callbackErr) || !errors.Is(err,toolsy.ErrStreamAborted) { t.Errorf("callback replaced: %v",err) }
 }) }
 wg.Wait()
 // Assert: one creation and one acknowledged cleanup/diagnostic per invocation, no retries.
 if creates.Load()!=calls || cancels.Load()!=calls || observed.Load()!=calls { t.Fatalf("counts create/cancel/observe=%d/%d/%d",creates.Load(),cancels.Load(),observed.Load()) }
}
func TestAcceptanceACooperativeCredentialBudget(t *testing.T) {
 // Arrange: opaque credential port cooperates with the cleanup deadline.
 exactErr:=errors.New("credentials stopped after budget")
 cred:=&acceptanceCredentials{fail:exactErr,wait:true}
 var requests atomic.Int32
 s:=httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter,r *http.Request){ requests.Add(1); w.WriteHeader(204) })); defer s.Close()
 var diag CancellationDiagnostic; var expired bool
 c:=mustClient(t,s.URL,WithAllowPrivateIPs(true),WithCancellationObserver(func(ctx context.Context,d CancellationDiagnostic){diag=d; expired=ctx.Err()==context.DeadlineExceeded}))
 ctx,cancel:=context.WithCancel(context.WithValue(t.Context(),acceptanceValueKey{},"preserved")); cancel()
 // Act.
 start:=time.Now(); c.cancelInterruptedTask(ctx,toolsy.NewRunEnv(nil,toolsy.WithCredentials(cred)),"t"); elapsed:=time.Since(start)
 // Assert: shared expired context, exact credential error, no request after failure.
 if elapsed<cancelTaskTimeout || elapsed>cancelTaskTimeout+time.Second { t.Fatalf("cleanup budget %v",elapsed) }
 if !expired || !cred.sawValue.Load() || diag.Cause!=exactErr || diag.Stage!=CancellationCredentials || diag.Acknowledged || requests.Load()!=0 {t.Fatalf("bad cooperative outcome: %+v expired=%v requests=%d",diag,expired,requests.Load())}
}
