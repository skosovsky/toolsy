package toolsy
import("context"; "encoding/json"; "errors"; "testing"; "github.com/stretchr/testify/require")
func TestR11AOverrideAndReplay(t *testing.T){
 // Arrange: typed value differs intentionally from its JSON wire override.
 tool,err:=NewTypedTool(TypedToolSpec[NoSubject,NoScope,struct{},int,string]{Name:"override",Description:"override",Options:[]ToolOption{WithIdempotent()},Handler:func(context.Context,TypedCallContext[NoSubject,NoScope],*RunEnv,ValidatedArgs[struct{}])(ToolResult[int,string],error){r:=NewToolResult[int,string](1);r.Raw=[]byte(`2`);r.RawMimeType=MimeTypeJSON;r.Audience=AudienceInternal;r.EnvelopeMetadata=map[string]any{"audit":"yes"};return r,nil}});require.NoError(t,err)
 cache,err:=NewResultCache(NewMemoryResultCacheStore(),constantPartition,JSONResultCodec[int,string]{},0);require.NoError(t,err)
 reg,err:=NewRegistryBuilder(WithExecutionProfile(cache)).Add(tool).Build();require.NoError(t,err)
 // Act and Assert: both representations, envelope and replay provenance survive.
 for i:=0;i<2;i++{var got Chunk;err=reg.Execute(t.Context(),ToolCall{ToolName:"override",Input:ToolInput{ArgsJSON:[]byte(`{}`)}},func(c Chunk)error{got=c;return nil});require.NoError(t,err);require.Equal(t,1,got.TypedResult);require.Equal(t,1,got.Envelope.Result);require.Equal(t,`2`,string(got.Data));require.Equal(t,AudienceInternal,got.Envelope.Audience);require.Equal(t,"yes",got.Envelope.Metadata["audit"]);if i==1{require.Equal(t,true,got.Envelope.Metadata[CacheReplayMetadata])}}
}

func TestR11ATypedEmptyPersistentAndUnencodable(t *testing.T){
 for _,noop:=range []bool{false,true}{t.Run("typed",func(t *testing.T){
 // Arrange: status omits wire while retaining typed host DTO.
 c:=Chunk{Event:EventResult,EmptyResult:!noop,Noop:noop,TypedResult:7};c.Envelope=NewResultEnvelope(7,nil,"",DeliveryClassText,AudienceInternal,map[string]any{"private":true})
 // Act.
 raw,err:=(JSONResultCodec[int,string]{}).EncodeResult(c);require.NoError(t,err);restored,err:=(JSONResultCodec[int,string]{}).DecodeResult(raw)
 // Assert.
 require.NoError(t,err);require.Equal(t,7,restored.TypedResult);require.Equal(t,7,restored.Envelope.Result);require.Empty(t,restored.Data);require.Equal(t,noop,restored.Noop);require.Equal(t,!noop,restored.EmptyResult)
 })}
 // Arrange: arbitrary BYOT Value that cannot serialize as JSON.
 tool,err:=NewTypedTool(TypedToolSpec[NoSubject,NoScope,struct{},map[string]any,string]{Name:"opaque",Description:"opaque",Handler:func(context.Context,TypedCallContext[NoSubject,NoScope],*RunEnv,ValidatedArgs[struct{}])(ToolResult[map[string]any,string],error){r:=NewToolResult[map[string]any,string](map[string]any{"callback":func(){}});r.Empty=true;return r,nil}});require.NoError(t,err)
 var got Chunk
 // Act.
 err=tool.Execute(t.Context(),nil,ToolInput{ArgsJSON:[]byte(`{}`)},func(c Chunk)error{got=c;return nil})
 // Assert: direct succeeds without serializing; JSON persistence explicitly cannot handle this host value.
 require.NoError(t,err);require.Empty(t,got.Data);require.IsType(t,func(){},got.TypedResult.(map[string]any)["callback"]);_,err=(JSONResultCodec[map[string]any,string]{}).EncodeResult(got);require.Error(t,err)
}
