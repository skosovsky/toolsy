package toolsygen

func TestR21BGenerateCompleteCleanupDiagnostic(t *testing.T) {
 // Arrange: existing generated target, new output and failed disposal of its old backup.
 dir:=t.TempDir()
 writeFile(t,filepath.Join(dir,"a.yaml"),"name: a\ndescription: a\nparameters:\n  type: object\n  properties: {}\n")
 cfg:=Config{Inputs:[]string{dir}}
 initial,err:=Generate(t.Context(),cfg);require.NoError(t,err);require.Len(t,initial.Files,1)
 target:=initial.Files[0]; writeFile(t,target,"package generated\n// old content\n")
 facade:=newDefaultOSFacade(defaultMaxGeneratorFileBytes);remove:=facade.remove;cause:=errors.New("disposal denied")
 facade.remove=func(path string)error{
  if strings.Contains(path,".bak-"){ data,e:=os.ReadFile(path);if e==nil&&len(data)>0{return cause} }
  return remove(path)
 }
 // Act.
 got,err:=generateWithFS(t.Context(),cfg,facade)
 // Assert: error cannot imply rollback, all installed files remain reported.
 require.ErrorIs(t,err,cause);require.Equal(t,initial.Files,got.Files)
 var diag *FileRecoveryError;require.ErrorAs(t,err,&diag);require.True(t,diag.CommitComplete)
 require.Equal(t,target,diag.Target);requireFileContent(t,diag.Backup,"package generated\n// old content\n")
 data,e:=os.ReadFile(target);require.NoError(t,e);require.Contains(t,string(data),"Code generated by toolsy-gen")
}

func TestR21BAggregatesAllRollbackCauses(t *testing.T){
 // Arrange: third backup fails after a and b installed; both restore operations fail independently.
 dir:=t.TempDir();a,b,c:=filepath.Join(dir,"a.go"),filepath.Join(dir,"b.go"),filepath.Join(dir,"c.go")
 for _,p:=range []string{a,b,c}{writeFile(t,p,"old:"+p)}
 primary,failA,failB:=errors.New("third backup failed"),errors.New("a restore failed"),errors.New("b restore failed")
 facade:=newDefaultOSFacade(defaultMaxGeneratorFileBytes);rename:=facade.rename
 facade.rename=func(old,new string)error{
  if old==c{return primary};if strings.Contains(old,".bak-"){if new==a{return failA};if new==b{return failB}}
  return rename(old,new)
 }
 g:=&generator{fs:facade}
 // Act.
 err:=g.commitFilesWithRollback(t.Context(),[]generatedFile{{Path:a,Content:[]byte("new")},{Path:b,Content:[]byte("new")},{Path:c,Content:[]byte("new")}},0o600)
 // Assert: both sole recovery copies remain and every error is reachable.
 for _,cause:=range []error{primary,failA,failB}{require.ErrorIs(t,err,cause)}
 for _,p:=range []string{a,b}{paths,e:=filepath.Glob(p+".bak-*");require.NoError(t,e);require.Len(t,paths,1);requireFileContent(t,paths[0],"old:"+p);require.Contains(t,err.Error(),paths[0])}
 requireFileContent(t,c,"old:"+c)
}
