package main
import("context";"fmt";"time"; t "github.com/skosovsky/toolsy")
func main(){ctx:=context.Background(); now:=time.Now().UTC(); b:=t.OperationBinding{Namespace:"n",Scope:"s",Subject:"u",OperationID:"i",Tool:"w",ManifestFingerprint:"m",PolicyFingerprint:"p",CanonicalDigest:"d",CanonicalRules:"c"}
for _,approved:=range []bool{false,true}{m:=t.NewMemoryOperationStore(); c:=t.OperationClaim{Binding:b,AttemptID:"a",RequiresApproval:approved,Issuer:"h",Now:now,LeaseUntil:now.Add(time.Minute)};if approved{c.GrantID="g"; must(m.PutGrant(ctx,t.ApprovalGrant{ID:"g",Issuer:"h",Binding:b,IssuedAt:now,ExpiresAt:now.Add(time.Hour),AllowRecovery:true}))};r,e:=m.Claim(ctx,c);must(e);if !r.Dispatch{panic("initial refused")}; must(m.Finish(ctx,t.OperationFinish{Binding:b,AttemptID:"a",State:t.OperationUnknown}));must(m.Resolve(ctx,t.OperationResolution{Binding:b,ExpectedAttemptID:"a",State:t.OperationRetryAuthorized,ProofID:"proof",Now:now}));var err error;m,err=t.RestoreOperationStore(m.Snapshot());must(err);c.AttemptID="b";r,e=m.Claim(ctx,c);must(e);if !r.Dispatch{panic("recovery refused")};must(m.Finish(ctx,t.OperationFinish{Binding:b,AttemptID:"b",State:t.OperationCompleted,Result:[]byte("receipt")}));m,err=t.RestoreOperationStore(m.Snapshot());must(err);record,found,e:=m.Inspect(ctx,b);must(e);if !found||len(record.Attempts)!=2||string(record.Result)!="receipt"{panic("recovery lost")};fmt.Printf("approved=%v recovery restored completed, attempts=%v\n",approved,record.Attempts)} }
func must(e error){if e!=nil{panic(e)}}
