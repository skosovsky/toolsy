package main
import("context";"encoding/json";"fmt";"os";"path/filepath";"time"; t "github.com/skosovsky/toolsy"; j "github.com/skosovsky/toolsy/adapters/execution/filejournal")
func main(){
 ctx:=context.Background(); now:=time.Now().UTC(); b:=t.OperationBinding{Namespace:"n",Scope:"s",Subject:"u",OperationID:"o",Tool:"t",ManifestFingerprint:"m",PolicyFingerprint:"p",CanonicalDigest:"d",CanonicalRules:"r"}; c:=t.OperationClaim{Binding:b,AttemptID:"a",GrantID:"g",RequiresApproval:false,Issuer:"host",Now:now,LeaseUntil:now.Add(time.Minute)}
 m:=t.NewMemoryOperationStore(); r,e:=m.Claim(ctx,c);must(e!=nil&&!r.Dispatch&&len(m.Snapshot().Records)==0,"invalid unused grant before mutation")
 dir,e:=os.MkdirTemp("/tmp","d14-a-final-journal-");if e!=nil{panic(e)}; defer os.RemoveAll(dir);path:=filepath.Join(dir,"journal.json"); f,e:=j.Open(path,0);if e!=nil{panic(e)};r,e=f.Claim(ctx,c);_,staterr:=os.Stat(path);must(e!=nil&&!r.Dispatch&&os.IsNotExist(staterr),"invalid initial journal claim leaves image absent")
 c.GrantID="";r,e=f.Claim(ctx,c);must(e==nil&&r.Dispatch,"valid unapproved dispatch"); e=f.Finish(ctx,t.OperationFinish{Binding:b,AttemptID:"a",State:t.OperationCompleted,Result:[]byte("known result")});must(e==nil,"valid unapproved finish");f,e=j.Open(path,0);must(e==nil,"reopen");rec,found,e:=f.Inspect(ctx,b);must(e==nil&&found&&rec.State==t.OperationCompleted,"completed restart");
 for _,approved:=range []bool{false,true}{
  m=t.NewMemoryOperationStore();c.GrantID="";c.RequiresApproval=approved;if approved{c.GrantID="g"};must(m.PutGrant(ctx,t.ApprovalGrant{ID:"g",Issuer:"host",Binding:b,IssuedAt:now.Add(-time.Minute),ExpiresAt:now.Add(time.Hour),AllowRecovery:true})==nil,"seed grant");r,e=m.Claim(ctx,c);must(e==nil&&r.Dispatch,"seed claim");must(m.Finish(ctx,t.OperationFinish{Binding:b,AttemptID:"a",State:t.OperationUnknown})==nil,"unknown");must(m.Resolve(ctx,t.OperationResolution{Binding:b,ExpectedAttemptID:"a",State:t.OperationRetryAuthorized,ProofID:"proof",Now:now.Add(time.Minute)})==nil,"resolve retry");before,_:=json.Marshal(m.Snapshot());c.AttemptID="switch";c.RequiresApproval=!approved;c.GrantID="";if c.RequiresApproval{c.GrantID="g"};r,e=m.Claim(ctx,c);after,_:=json.Marshal(m.Snapshot());must(e!=nil&&!r.Dispatch&&string(before)==string(after),"approval-mode switch rejected immutable");_,e=t.RestoreOperationStore(m.Snapshot());must(e==nil,"post-reject restore");c.RequiresApproval=approved;c.GrantID="";if approved{c.GrantID="g"};r,e=m.Claim(ctx,c);must(e==nil&&r.Dispatch,"same-mode recovery");_,e=t.RestoreOperationStore(m.Snapshot());must(e==nil,"post-recovery restore"); c.AttemptID="a"
 }
 fmt.Println("independent memory/durable probes PASS: invalid claims before dispatch/mutation, valid unapproved finish/reopen, approval upgrades/downgrades rejected, same-mode recovery restores")
}
