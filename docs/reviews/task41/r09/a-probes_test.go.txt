package toolsy
import("errors";"fmt";"sync";"testing";"time")
type acceptanceACodec struct{ encode func(); decode func() error }
func(c acceptanceACodec)Encode(v int)([]byte,error){if c.encode!=nil {c.encode()};return []byte(fmt.Sprint(v)),nil}
func(c acceptanceACodec)Decode([]byte)(int,error){if c.decode!=nil{return 7,c.decode()};return 7,nil}
func TestAcceptanceAConcurrentFreeze(t *testing.T){
 for round:=0;round<30;round++ {
  r:=NewStateCodecRegistry();start:=make(chan struct{});out:=make(chan struct{key string;err error},200);var wg sync.WaitGroup
  for worker:=0;worker<10;worker++ {wg.Add(1);go func(worker int){defer wg.Done();<-start;for n:=0;n<20;n++ {key:=fmt.Sprintf("%d/%d",worker,n);err:=RegisterJSONCodec[int](r,key,WithStateSlotRequired());out<-struct{key string;err error}{key,err}}}(worker)}
  close(start);s,err:=NewSession(nil,WithStateCodecRegistry(r),WithStrictStateCodecs(true));if err!=nil{t.Fatal(err)};wg.Wait();close(out)
  for result:=range out {if result.err==nil{SetSessionState(s,result.key,7)}else if !errors.Is(result.err,ErrStateCodecRegistryFrozen){t.Fatal(result.err)}}
  cp,err:=s.ExportCheckpoint();if err!=nil{t.Fatal(err)};restored,err:=NewSessionFromCheckpoint(nil,cp,WithStateCodecRegistry(r),WithStrictStateCodecs(true));if err!=nil{t.Fatal(err)}
  if restored.Binding().StateSchemaDigest!=cp.Binding.StateSchemaDigest{t.Fatal("schema drift")};if err:=r.RegisterFromPrototype("late",0);!errors.Is(err,ErrStateCodecRegistryFrozen){t.Fatal(err)}
 }
}
