func fixtureGit(t *testing.T, dir string, args ...string) string {
	t.Helper()
	cmd := exec.Command("git", args...)
	cmd.Dir = dir
	cmd.Env = append(commandEnvironment(false), "GIT_CONFIG_COUNT=0")
	raw, err := cmd.CombinedOutput()
	if err != nil {
		t.Fatalf("git %v: %v\n%s", args, err, raw)
	}
	return string(bytes.TrimSpace(raw))
}

func writeFixture(t *testing.T, path, text string) {
	t.Helper()
	if err := os.MkdirAll(filepath.Dir(path), 0o755); err != nil {
		t.Fatal(err)
	}
	if err := os.WriteFile(path, []byte(text), 0o644); err != nil {
		t.Fatal(err)
	}
}

func releaseFixture(t *testing.T) (string, string) {
	t.Helper()
	parent := t.TempDir()
	source := filepath.Join(parent, "source")
	remote := filepath.Join(parent, "remote.git")
	if err := os.Mkdir(source, 0o755); err != nil {
		t.Fatal(err)
	}
	fixtureGit(t, parent, "init", "--bare", remote)
	fixtureGit(t, source, "init", "-b", "main")
	fixtureGit(t, source, "config", "user.name", "Release fixture")
	fixtureGit(t, source, "config", "user.email", "fixture@example.invalid")
	fixtureGit(t, source, "config", "commit.gpgsign", "false")
	writeFixture(t, filepath.Join(source, "go.mod"), "module example.invalid/fixture\n\ngo 1.27.1\n")
	writeFixture(t, filepath.Join(source, "value.go"), "package fixture\nconst Value = 1\n")
	writeFixture(
		t,
		filepath.Join(source, "sub/go.mod"),
		"module example.invalid/fixture/sub\n\ngo 1.27.1\n\nrequire example.invalid/fixture v0.0.0\nreplace example.invalid/fixture => ../\n",
	)
	writeFixture(
		t,
		filepath.Join(source, "sub/value.go"),
		"package sub\nimport root \"example.invalid/fixture\"\nconst Value = root.Value\n",
	)
	fixtureGit(t, source, "add", "go.mod", "value.go", "sub/go.mod", "sub/value.go")
	fixtureGit(t, source, "commit", "-m", "fixture")
	fixtureGit(t, source, "tag", "v0.1.0")
	fixtureGit(t, source, "remote", "add", "origin", remote)
	fixtureGit(t, source, "push", "origin", "main", "--tags")
	return source, remote
}

func releaseScript(t *testing.T) string {
	t.Helper()
	if path := os.Getenv("TOOLSY_RELEASE_SCRIPT"); path != "" {
		return path
	}
	// Commit the current runner into a disposable bootstrap repository. The
	// working repository has intentionally uncommitted implementation changes.
	root, rootErr := filepath.Abs("../..")
	if rootErr != nil {
		t.Fatal(rootErr)
	}
	bootstrap := t.TempDir()
	for _, dir := range []string{"cmd/toolsy-release", "internal/release", "scripts"} {
		entries, err := os.ReadDir(filepath.Join(root, dir))
		if err != nil {
			t.Fatal(err)
		}
		for _, entry := range entries {
			name := entry.Name()
			if entry.IsDir() || strings.HasSuffix(name, "_test.go") || (dir == "scripts" && name != "release.sh") {
				continue
			}
			data, readErr := os.ReadFile(filepath.Join(root, dir, name))
			if readErr != nil {
				t.Fatal(readErr)
			}
			if dir == "scripts" && name == "release.sh" { data = []byte("#!/bin/bash\nset -euo pipefail\n# The asynchronous bootstrap build owns a process group on supported hosts.\nset -m\n\nunset GIT_DIR GIT_WORK_TREE GIT_INDEX_FILE GIT_OBJECT_DIRECTORY GIT_ALTERNATE_OBJECT_DIRECTORIES GIT_COMMON_DIR GIT_NAMESPACE\nif [ -n \"${GIT_CONFIG_PARAMETERS:-}\" ]; then\n    echo \"release does not support GIT_CONFIG_PARAMETERS; use Git config\" >&2\n    exit 1\nfi\nif [ \"${GIT_CONFIG_COUNT:-0}\" != 0 ]; then\n    echo \"bootstrap does not support counted Git config; use Git config or the native CLI\" >&2\n    exit 1\nfi\n\nscript_root=$(CDPATH= cd -- \"$(dirname -- \"$0\")/..\" && pwd)\nrunner_dir=$(mktemp -d \"${TMPDIR:-/tmp}/toolsy-release-runner.XXXXXX\")\nchild_pid=\nchild_phase=\ncleanup() {\n    if [ -n \"$child_pid\" ]; then\n        if [ \"$child_phase\" = build ]; then\n            kill -TERM -- \"-$child_pid\" 2>/dev/null || true\n            for attempt in {1..20}; do\n                kill -0 \"$child_pid\" 2>/dev/null || break\n                sleep 0.1\n            done\n            kill -KILL -- \"-$child_pid\" 2>/dev/null || true\n        else\n            kill -TERM \"$child_pid\" 2>/dev/null || true\n        fi\n        wait \"$child_pid\" 2>/dev/null || true\n    fi\n    rm -rf -- \"$runner_dir\"\n}\ntrap cleanup EXIT\ntrap 'exit 130' INT\ntrap 'exit 143' TERM HUP\n\n# Bootstrap only committed code; untracked Go files must not enter the runner.\nif [ -n \"$(GIT_OPTIONAL_LOCKS=0 git -C \"$script_root\" -c core.fsmonitor=false status --porcelain --untracked-files=no)\" ]; then\n    echo \"commit or stash tracked changes before release\" >&2\n    exit 1\nfi\n# Archive transformations must be rejected before any runner code is built.\nGIT_OPTIONAL_LOCKS=0 git -C \"$script_root\" ls-tree -r -z --name-only HEAD |\n    git -C \"$script_root\" check-attr --cached -z --stdin filter working-tree-encoding export-ignore export-subst ident crlf text eol |\n    while IFS= read -r -d '' attribute_path; do\n        IFS= read -r -d '' attribute_name\n        IFS= read -r -d '' attribute_value\n        if [ \"$attribute_value\" != unspecified ] && [ \"$attribute_value\" != unset ]; then\n            echo \"release does not support transforming Git attribute $attribute_name=$attribute_value on $attribute_path\" >&2\n            exit 1\n        fi\n    done\nmkdir \"$runner_dir/source\"\nGIT_OPTIONAL_LOCKS=0 git -C \"$script_root\" -c core.autocrlf=input -c core.eol=lf archive HEAD | tar -x -C \"$runner_dir/source\"\nGOWORK=off go build -C \"$runner_dir/source\" -mod=readonly -buildvcs=false -o \"$runner_dir/toolsy-release\" ./cmd/toolsy-release &\nchild_pid=$!\nchild_phase=build\nwait \"$child_pid\"\nchild_pid=\n# Keep the CLI in the terminal's foreground process group for its prompt.\nset +m\n\"$runner_dir/toolsy-release\" \"$@\" <&0 &\nchild_pid=$!\nchild_phase=release\nwait \"$child_pid\"\nchild_pid=\n") }
			writeFixture(t, filepath.Join(bootstrap, dir, name), string(data))
		}
	}
	writeFixture(
		t,
		filepath.Join(bootstrap, "go.mod"),
		"module github.com/skosovsky/toolsy\n\ngo 1.27.1\n\nrequire golang.org/x/mod v0.41.0\n",
	)
	sum, err := os.ReadFile(filepath.Join(root, "go.sum"))
	if err != nil {
		t.Fatal(err)
	}
	writeFixture(t, filepath.Join(bootstrap, "go.sum"), string(sum))
	fixtureGit(t, bootstrap, "init", "-b", "main")
	fixtureGit(t, bootstrap, "config", "commit.gpgsign", "false")
	fixtureGit(t, bootstrap, "config", "user.name", "Fixture")
	fixtureGit(t, bootstrap, "config", "user.email", "fixture@example.invalid")
	fixtureGit(t, bootstrap, "add", ".")
	fixtureGit(t, bootstrap, "commit", "-m", "bootstrap fixture")
	// A source-working-tree build would fail on this untracked file.
	writeFixture(t, filepath.Join(bootstrap, "cmd/toolsy-release/untracked.go"), "INVALID_UNTRACKED_GO\n")
	return filepath.Join(bootstrap, "scripts/release.sh")
}

func TestReleasePreservesUntrackedSource(t *testing.T) {
	// Arrange: only committed files may become release inputs.
	source, remote := releaseFixture(t)
	note := filepath.Join(source, "private-note.txt")
	writeFixture(t, note, "FAKE_LOCAL_MARKER\n")
	beforeIndex, err := os.ReadFile(filepath.Join(source, ".git/index"))
	if err != nil {
		t.Fatal(err)
	}
	beforeHead := fixtureGit(t, source, "rev-parse", "HEAD")
	ctx, cancel := context.WithTimeout(context.Background(), 2*time.Minute)
	defer cancel()
	cmd := exec.CommandContext(ctx, "bash", releaseScript(t), "patch", ". ./sub")
	cmd.Dir = source
	cmd.Env = append(commandEnvironment(false), "GIT_CONFIG_COUNT=0")
	cmd.Stdin = bytes.NewBufferString("y\n")
	// Act.
	output, runErr := cmd.CombinedOutput()
	// Assert.
	if runErr != nil {
		t.Fatalf("release: %v\n%s", runErr, output)
	}
	content, noteErr := os.ReadFile(note)
	if noteErr != nil || string(content) != "FAKE_LOCAL_MARKER\n" {
		t.Errorf("source note lost/changed: %v %q", noteErr, content)
	}
	if head := fixtureGit(t, source, "rev-parse", "HEAD"); head != beforeHead {
		t.Errorf("source HEAD changed: %s", head)
	}
	if branch := fixtureGit(t, source, "branch", "--show-current"); branch != "main" {
		t.Errorf("source branch changed: %q", branch)
	}
	afterIndex, indexErr := os.ReadFile(filepath.Join(source, ".git/index"))
	if indexErr != nil || !bytes.Equal(beforeIndex, afterIndex) {
		t.Error("source index changed")
	}
	show := exec.Command("git", "--git-dir", remote, "show", "v0.1.1:private-note.txt")
	if raw, showErr := show.CombinedOutput(); showErr == nil {
		t.Errorf("private note published: %s", raw)
	}
}
