func TestFinalAExactAdversarial(t *testing.T) {
 // Arrange: adjacent exact numbers, escaped duplicate names, nested duplicates and fractional integer.
 calls:=0
 schema:=toolsy.MapSchemaProvider{"type":"object","properties":map[string]any{"id":map[string]any{"type":"integer","minimum":json.Number("9007199254740993"),"maximum":json.Number("9223372036854775807")}},"required":[]any{"id"}}
 tool,err:=toolsy.NewDynamicToolFromSpec(toolsy.DynamicToolSpec{Name:"a_numeric",Description:"exact",Schema:schema,Handler:func(_ context.Context,_ *toolsy.RunEnv,a map[string]any,_ func(toolsy.Chunk)error)error{calls++;if _,ok:=a["id"].(json.Number);!ok{t.Fatalf("numeric type %T",a["id"])};return nil}})
 if err!=nil{t.Fatal(err)}
 tests:=[]struct{raw string; valid bool}{{`{"id":9007199254740993}`,true},{`{"id":9223372036854775807}`,true},{`{"id":9007199254740992}`,false},{`{"id":9223372036854775808}`,false},{`{"id":9007199254740993.1}`,false},{`{"id":9007199254740993,"\u0069d":9007199254740994}`,false},{`{"id":9007199254740993,"extra":{"n":1,"n":2}}`,false},{`{"id":9007199254740993} {}`,false},{`{"id":9007199254740993,"x":`+strings.Repeat("[",130)+`0`+strings.Repeat("]",130)+`}`,false}}
 // Act and assert: exact validation before handler dispatch.
 for _,tc:=range tests{e:=tool.Execute(context.Background(),toolsy.NewRunEnv(nil),toolsy.ToolInput{ArgsJSON:[]byte(tc.raw)},func(toolsy.Chunk)error{return nil});if (e==nil)!=tc.valid{t.Fatalf("raw=%s expectedvalid=%t err=%v",tc.raw,tc.valid,e)}}
 if calls!=2{t.Fatalf("dispatches=%d",calls)}
}
func TestFinalAKnownPhaseRejectsMisleadingCauses(t *testing.T){
 for _,cause:=range []error{context.DeadlineExceeded,toolsy.ErrStreamAborted,toolsy.NewValidationError("repair"),errors.New("plain")}{
 // Arrange: effect is complete, host supplied callback pretends interruption/input validation.
 effects,results:=0,0
 tool,err:=toolsy.NewTypedTool(toolsy.TypedToolSpec[toolsy.NoSubject,toolsy.NoScope,struct{},string,string]{Name:"a_write",Description:"write",Handler:func(context.Context,toolsy.TypedCallContext[toolsy.NoSubject,toolsy.NoScope],*toolsy.RunEnv,toolsy.ValidatedArgs[struct{}])(toolsy.ToolResult[string,string],error){effects++;return toolsy.NewToolResult[string,string]("ok"),nil},ResultValidator:func(string)error{return cause}})
 if err!=nil{t.Fatal(err)}
 // Act.
 err=tool.Execute(context.Background(),toolsy.NewRunEnv(nil),toolsy.ToolInput{ArgsJSON:[]byte(`{}`)},func(toolsy.Chunk)error{results++;return nil})
 // Assert: cause inspectable while outer phase never grants another dispatch.
 te,ok:=toolsy.AsToolError(err);var ce *toolsy.ResultContractError
 if !ok||te.Code!=toolsy.CodeInternal||te.Retryable||toolsy.ClientCorrectable(te.Code)||!errors.As(err,&ce)||ce.Kind!="result_validator"||!errors.Is(err,cause)||effects!=1||results!=0{t.Fatalf("phase err=%v effects=%d results=%d",err,effects,results)}
 }
}
