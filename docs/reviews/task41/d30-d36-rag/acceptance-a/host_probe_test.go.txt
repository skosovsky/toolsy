package main
import("context";"errors";"testing";"github.com/skosovsky/toolsy/toolkits/rag")
func TestAcceptancePolicyCancelFalsePreservesCause(t *testing.T){
 // Arrange: a host predicate cancels request while denying fallback on primary failure.
 ctx,cancel:=context.WithCancelCause(t.Context());cause:=errors.New("policy canceled");primaryErr:=errors.New("primary denied");next:=0
 k,err:=newKnowledgeBase(func(context.Context,string)([]rag.Document,error){return nil,primaryErr},func(context.Context,string)([]rag.Document,error){next++;return nil,nil},func(context.Context,string)([]rag.Document,error){next++;return nil,nil},func([]rag.Document,error)bool{cancel(cause);return false});if err!=nil{t.Fatal(err)}
 // Act.
 _,err=k.Retrieve(ctx,"x")
 // Assert: cancellation is terminal; custom parent cause survives policy callback.
 if next!=0||!errors.Is(err,context.Canceled)||!errors.Is(err,cause){t.Fatalf("lost terminal policy cancellation: got=%v next=%d ctx=%v cause=%v",err,next,ctx.Err(),context.Cause(ctx))}
}

func TestAcceptanceHostErrorsAndIdentity(t *testing.T){
 // Arrange: a host endpoint returns cancellation with a live parent; predicate must never see it.
 for _,terminal:=range []error{context.Canceled,context.DeadlineExceeded}{calls:=0;k,e:=newKnowledgeBase(func(context.Context,string)([]rag.Document,error){return nil,terminal},func(context.Context,string)([]rag.Document,error){calls++;return nil,nil},func(context.Context,string)([]rag.Document,error){calls++;return nil,nil},func([]rag.Document,error)bool{calls++;return true});if e!=nil{t.Fatal(e)}
 // Act/Assert.
 _,e=k.Retrieve(t.Context(),"x");if !errors.Is(e,terminal)||calls!=0{t.Fatalf("terminal retried: %v calls%d",e,calls)}}
 // Arrange: public identity scopes chunk to its source and leaves unidentified units intact.
 docs:=[]rag.Document{{ID:"a",SourceURI:"doc://1"},{ID:"a",SourceURI:"doc://2"},{ID:"b",SourceURI:"doc://1"},{ID:"a",SourceURI:"doc://1"},{Content:"unknown"},{Content:"unknown"}}
 // Act/Assert.
 out:=uniquePublicChunks(docs);if len(out)!=5||out[1].SourceURI!="doc://2"||out[2].ID!="b"{t.Fatalf("identity lost %#v",out)}
}
