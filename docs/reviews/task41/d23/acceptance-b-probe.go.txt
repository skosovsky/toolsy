type discoveryFixture struct {
	reflectionpb.UnimplementedServerReflectionServer

	names     []string
	blobs     [][]byte
	excluded  atomic.Int32
	requested atomic.Int32
	listed    atomic.Int32
}

func (s *discoveryFixture) listResponse() *reflectionpb.ServerReflectionResponse {
	entries := make([]*reflectionpb.ServiceResponse, len(s.names))
	for i, name := range s.names {
		entries[i] = &reflectionpb.ServiceResponse{Name: name}
	}
	return &reflectionpb.ServerReflectionResponse{
		MessageResponse: &reflectionpb.ServerReflectionResponse_ListServicesResponse{
			ListServicesResponse: &reflectionpb.ListServiceResponse{Service: entries}},
	}
}

func (s *discoveryFixture) fileResponse() *reflectionpb.ServerReflectionResponse {
	return &reflectionpb.ServerReflectionResponse{
		MessageResponse: &reflectionpb.ServerReflectionResponse_FileDescriptorResponse{
			FileDescriptorResponse: &reflectionpb.FileDescriptorResponse{FileDescriptorProto: s.blobs}},
	}
}

func (s *discoveryFixture) ServerReflectionInfo(
	stream grpc.BidiStreamingServer[reflectionpb.ServerReflectionRequest, reflectionpb.ServerReflectionResponse],
) error {
	for {
		req, err := stream.Recv()
		if errors.Is(err, io.EOF) {
			return nil
		}
		if err != nil {
			return err
		}
		response := s.listResponse()
		if req.GetFileContainingSymbol() == "" {
			s.listed.Add(1)
		}
		if symbol := req.GetFileContainingSymbol(); symbol != "" {
			s.requested.Add(1)
			if symbol != "fixture.Service" {
				s.excluded.Add(1)
				response = &reflectionpb.ServerReflectionResponse{
					MessageResponse: &reflectionpb.ServerReflectionResponse_ErrorResponse{
						ErrorResponse: &reflectionpb.ErrorResponse{
							ErrorCode:    5,
							ErrorMessage: "excluded service deliberately broken",
						},
					},
				}
			} else {
				response = s.fileResponse()
			}
		}
		if err = stream.Send(response); err != nil {
			return err
		}
	}
}

func newDiscoveryFixture(t *testing.T, names []string) (*discoveryFixture, *grpc.ClientConn) {
	t.Helper()
	raw, err := proto.Marshal(protodesc.ToFileDescriptorProto(fixture(t, false, false)))
	if err != nil {
		t.Fatal(err)
	}
	scripted := &discoveryFixture{
		UnimplementedServerReflectionServer: reflectionpb.UnimplementedServerReflectionServer{},
		names:                               names,
		blobs:                               [][]byte{raw},
		excluded:                            atomic.Int32{},
		requested:                           atomic.Int32{},
	}
	listener := bufconn.Listen(testBufconnSize)
	server := grpc.NewServer()
	reflectionpb.RegisterServerReflectionServer(server, scripted)
	go func() { _ = server.Serve(listener) }()
	t.Cleanup(server.Stop)
	conn, err := grpc.NewClient(
		"passthrough:///bufconn",
		grpc.WithTransportCredentials(insecure.NewCredentials()),
		grpc.WithContextDialer(func(context.Context, string) (net.Conn, error) { return listener.Dial() }),
	)
	if err != nil {
		t.Fatal(err)
	}
	t.Cleanup(func() { _ = conn.Close() })
	return scripted, conn
}

func TestReflectFiltersBeforeDescriptorRequests(t *testing.T) {
	// Arrange: the excluded service fails reflection if requested.
	server, conn := newDiscoveryFixture(t, []string{"broken.Service", "fixture.Service", "fixture.Service"})
	// Act.
	tools, err := Reflect(t.Context(), conn, Options{Services: []string{"fixture.Service"}})
	// Assert: excluded failures never affect selected discovery; duplicate names fetch once.
	if err != nil || len(tools) != 1 {
		t.Fatalf("selected discovery: tools=%d err=%v", len(tools), err)
	}
	if server.excluded.Load() != 0 || server.requested.Load() != 1 {
		t.Fatalf("excluded=%d requested=%d", server.excluded.Load(), server.requested.Load())
	}
}

func TestReflectInclusiveDiscoveryLimits(t *testing.T) {
	for _, kind := range []string{"services", "files", "bytes"} {
		for _, exact := range []bool{true, false} {
			t.Run(kind+"/"+map[bool]string{true: "exact", false: "over"}[exact], func(t *testing.T) {
				// Arrange: exclusions/duplicates count even though one method is published.
				server, conn := newDiscoveryFixture(t, []string{"broken.Service", "fixture.Service", "fixture.Service"})
				server.blobs = append(server.blobs, server.blobs[0])
				opts := discoveryLimitOptions(server, kind, exact)
				// Act.
				tools, err := Reflect(t.Context(), conn, opts)
				// Assert: inclusive bound succeeds; +1 work refuses the entire snapshot.
				assertDiscoveryLimitResult(t, kind, exact, server, len(tools), err)
			})
		}
	}
}

func TestReflectRejectsNegativeLimitsBeforeRPC(t *testing.T) {
	for _, opts := range []Options{{MaxDiscoveryServices: -1}, {MaxDescriptorFiles: -1}, {MaxDiscoveryBytes: -1}, {MaxResponseBytes: -1}} {
		// Arrange.
		server, conn := newDiscoveryFixture(t, []string{"fixture.Service"})
		// Act.
		tools, err := Reflect(t.Context(), conn, opts)
		// Assert: configuration refusal happens before opening/sending reflection RPCs.
		if err == nil || len(tools) != 0 || server.listed.Load() != 0 {
			t.Fatalf("negative options: tools=%d err=%v listed=%d", len(tools), err, server.listed.Load())
		}
	}
}

func TestReflectSelectedDependenciesAndConflicts(t *testing.T) {
	for _, conflict := range []bool{false, true} {
		// Arrange: selected service depends on an ordinary imported message.
		server, conn := newDiscoveryFixture(t, []string{"fixture.Service"})
		root := protodesc.ToFileDescriptorProto(fixture(t, false, false))
		dependency := &descriptorpb.FileDescriptorProto{
			Name:    new("dependency.proto"),
			Syntax:  new("proto3"),
			Package: new("dep"),
			MessageType: []*descriptorpb.DescriptorProto{
				{
					Name: new("Payload"),
					Field: []*descriptorpb.FieldDescriptorProto{
						{
							Name:   new("text"),
							Number: new(int32(1)),
							Type:   descriptorpb.FieldDescriptorProto_TYPE_STRING.Enum(),
						},
					},
				},
			},
		}
		root.Dependency = []string{"dependency.proto"}
		root.MessageType[0].Field = append(
			root.MessageType[0].Field,
			&descriptorpb.FieldDescriptorProto{
				Name:     new("payload"),
				Number:   new(int32(3)),
				Type:     descriptorpb.FieldDescriptorProto_TYPE_MESSAGE.Enum(),
				TypeName: new(".dep.Payload"),
			},
		)
		rawRoot, err := proto.Marshal(root)
		if err != nil {
			t.Fatal(err)
		}
		rawDependency, err := proto.Marshal(dependency)
		if err != nil {
			t.Fatal(err)
		}
		server.blobs = [][]byte{rawRoot, rawDependency}
		if conflict {
			dependency.Package = new("forged")
			changed, marshalErr := proto.Marshal(dependency)
			if marshalErr != nil {
				t.Fatal(marshalErr)
			}
			server.blobs = append(server.blobs, changed)
		}
		// Act.
		tools, err := Reflect(t.Context(), conn, Options{Services: []string{"fixture.Service"}})
		// Assert: dependencies compile with selected service; conflicting same-name files fail atomically.
		if conflict {
			if err == nil || len(tools) != 0 {
				t.Fatalf("conflict: tools=%d err=%v", len(tools), err)
			}
		} else {
			if err != nil || len(tools) != 1 {
				t.Fatalf("dependency: tools=%d err=%v", len(tools), err)
			}
		}
	}
}

func TestReflectCancellationAndPerMessageBound(t *testing.T) {
	// Arrange: caller owns deadline and connection; receive cap acts before aggregate processing.
	_, conn := newDiscoveryFixture(t, []string{"fixture.Service"})
	ctx, cancel := context.WithCancel(t.Context())
	cancel()
	// Act/Assert.
	tools, err := Reflect(ctx, conn, Options{})
	if !errors.Is(err, context.Canceled) && status.Code(err) != codes.Canceled {
		t.Fatal("cancelled discovery succeeded")
	}
	if len(tools) != 0 {
		t.Fatal("cancelled discovery published tools")
	}
	tools, err = Reflect(t.Context(), conn, Options{MaxDiscoveryBytes: 1})
	if err == nil || len(tools) != 0 {
		t.Fatalf("receive bound: tools=%d err=%v", len(tools), err)
	}
	// The adapter must not close the borrowed client connection on either failure.
	tools, err = Reflect(t.Context(), conn, Options{})
	if err != nil || len(tools) != 1 {
		t.Fatalf("connection lost: tools=%d err=%v", len(tools), err)
	}
}

func discoveryLimitOptions(server *discoveryFixture, kind string, exact bool) Options {
	difference := 0
	if !exact {
		difference = 1
	}
	opts := Options{Services: []string{"fixture.Service"}}
	switch kind {
	case "services":
		opts.MaxDiscoveryServices = len(server.names) - difference
	case "files":
		opts.MaxDescriptorFiles = len(server.blobs) - difference
	case "bytes":
		opts.MaxDiscoveryBytes = proto.Size(server.listResponse()) + proto.Size(server.fileResponse()) - difference
	}
	return opts
}

func assertDiscoveryLimitResult(t *testing.T, kind string, exact bool, server *discoveryFixture, count int, err error) {
	t.Helper()
	if exact {
		if err != nil || count != 1 {
			t.Fatalf("exact: tools=%d err=%v", count, err)
		}
		return
	}
	if !errors.Is(err, ErrDiscoveryLimit) || count != 0 {
		t.Fatalf("over: tools=%d err=%v", count, err)
	}
	if kind == "services" && server.requested.Load() != 0 {
		t.Fatal("fetched descriptors after service cap")
	}
}


type reviewBOwnedStreamFixture struct {
 reflectionpb.UnimplementedServerReflectionServer
 inner *discoveryFixture
 finished chan struct{}
}
func (s *reviewBOwnedStreamFixture) ServerReflectionInfo(stream grpc.BidiStreamingServer[reflectionpb.ServerReflectionRequest, reflectionpb.ServerReflectionResponse]) error {
 defer close(s.finished)
 for i:=0;i<2;i++ {
  req,err:=stream.Recv(); if err!=nil{return err}
  resp:=s.inner.listResponse(); if req.GetFileContainingSymbol()!="" {resp=s.inner.fileResponse()}
  if err=stream.Send(resp);err!=nil{return err}
 }
 // Ignore send-side EOF; only the adapter's owned context cancellation releases this server.
 <-stream.Context().Done()
 return stream.Context().Err()
}

 // Arrange: host, original request, and unknown bytes must all count.
 response:=&reflectionpb.ServerReflectionResponse{ValidHost:"extra-envelope", OriginalRequest:&reflectionpb.ServerReflectionRequest{Host:"host",MessageRequest:&reflectionpb.ServerReflectionRequest_ListServices{ListServices:""}}}

 if err=budget.response(response);err!=nil{t.Fatal(err)}

}
