func TestProviderContextEnvAndNilPriority(t *testing.T){
 for _,idx:=range []int{0,1}{
 // Arrange: provider observes exact caller context/env and owns cooperative cancellation.
 ctx,cancel:=context.WithCancel(t.Context());cancel();env:=toolsy.NewRunEnv(nil);calls:=0
 opts:=[]timetool.Option{timetool.WithLocation(time.UTC),timetool.WithLocationProvider(func(got context.Context,run *toolsy.RunEnv)(*time.Location,error){calls++;if got!=ctx||run!=env{t.Fatal("callback identity lost")};return nil,got.Err()})}
 // Act.
 out,err:=execute(t,opts,idx,ctx,env)
 // Assert: either framework prechecks cancellation or callback cooperates; no output/fallback.
 if !errors.Is(err,context.Canceled)||out!=nil{t.Fatalf("%v %v",out,err)};if calls>1{t.Fatal(calls)}
 // Arrange/Act: explicit nil returned zone must fail even with static UTC.
 out,err=execute(t,[]timetool.Option{timetool.WithLocation(time.UTC),timetool.WithLocationProvider(func(context.Context,*toolsy.RunEnv)(*time.Location,error){return nil,nil})},idx,t.Context(),env)
 // Assert.
 if !errors.Is(err,toolsy.ErrValidation)||out!=nil{t.Fatalf("%v %v",out,err)}
 }
