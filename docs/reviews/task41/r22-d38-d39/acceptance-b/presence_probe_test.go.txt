package main
import("context";"encoding/json";"testing";"bytes";"github.com/stretchr/testify/require";"github.com/skosovsky/toolsy")
func TestAcceptanceBRepresentation(t *testing.T){
 // Arrange.
 h:=&captureHandler{}; tool,err:=NewPresenceTool(h);require.NoError(t,err)
 raw:=[]byte(`{ "r_string":"", "r_integer":1e30,"r_boolean":false,"r_datetime":"not-a-date","r_datetimes":["also-not-date"],"r_strings":[],"r_integers":[900719925474099300000,1.0,2e2],"r_booleans":[false],"r_nullable_string":null,"r_nullable_integer":1e30,"r_nullable_boolean":false,"r_nullable_strings":[],"r_nullable_integers":[ 1.0, 9e20 ],"r_nullable_booleans":[false],"o_strings":[],"o_datetimes":[""],"unknown":{"value":1e30}}`)
 // Act.
 err=tool.Execute(context.Background(),toolsy.NewRunEnv(nil),toolsy.ToolInput{ArgsJSON:raw},func(toolsy.Chunk)error{return nil})
 // Assert.
 require.NoError(t,err);require.Equal(t,1,h.calls);require.True(t,bytes.Equal(raw,h.input.RawJSON));require.Equal(t,"1e30",h.input.RInteger.String()); require.Nil(t,h.input.OString);require.Equal(t,[]json.Number{"900719925474099300000","1.0","2e2"},*h.input.RIntegers);require.Equal(t,[]string{"also-not-date"},*h.input.RDatetimes);require.Equal(t,[]string{""},h.input.ODatetimes)
 _,err=h.input.RInteger.Int64();require.Error(t,err)
 require.Equal(t,"[1.0,9e20]",string(h.input.RNullableIntegers))
 encoded,err:=json.Marshal(h.input);require.NoError(t,err);var snapshot map[string]json.RawMessage;require.NoError(t,json.Unmarshal(encoded,&snapshot));require.NotContains(t,snapshot,"unknown");require.NotContains(t,snapshot,"o_strings");require.Equal(t,"null",string(snapshot["o_nullable_string"]));require.NotEqual(t,string(raw),string(encoded))
}
func TestAcceptanceBMalformedAndRootNoDispatch(t *testing.T){
 for _,raw:=range []string{"null","[]","\"x\"",`{"r_integer":1.5}`,`{"r_integer":true}`} {
 // Arrange.
 h:=&captureHandler{}; tool,err:=NewPresenceTool(h);require.NoError(t,err)
 // Act.
 err=tool.Execute(context.Background(),toolsy.NewRunEnv(nil),toolsy.ToolInput{ArgsJSON:[]byte(raw)},func(toolsy.Chunk)error{return nil})
 // Assert.
 require.ErrorIs(t,err,toolsy.ErrValidation);require.Zero(t,h.calls)
 }
}
