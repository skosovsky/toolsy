func TestR13BRealTTLAndReadmission(t *testing.T) {
 // Arrange.
 c, tr := retirementClient(t, WithSubscriptionLimits(SubscriptionLimits{MaxActive:2,MaxTracked:2,RetireTTL:20*time.Millisecond}))

 b, err := c.Listen(t.Context(), SubscriptionFilter{ToolsListChanged:true}); require.NoError(t,err)
 defer b.Close()
 // Act/Assert: retained A denies admission, then real TTL expiration frees it.

 select {case <-b.Events: case <-time.After(time.Second): t.Fatal("B lost its route")}
 require.Equal(t,uint64(1),c.DiscoveryGeneration(InvalidationTools))
}

type r13BPublicationPort struct {
 *bindingContractTransport
 entered chan struct{}
 resume chan struct{}
 block bool
}
func(p *r13BPublicationPort) ReplaceToolHeaderBindings(bindings map[string][]HTTPToolHeaderBinding)error{
 if p.block {close(p.entered); <-p.resume}
 return p.bindingContractTransport.ReplaceToolHeaderBindings(bindings)
}
func TestR13BCancellationAfterPublicationCommitCompletes(t *testing.T){
 // Arrange: full discovery blocks inside the trusted publication facet.
 name:="old"
 base:=newContractTransport(func(method string,_ json.RawMessage)(json.RawMessage,error){
 if method==MethodServerDiscover{return completeDiscovery(ServerCapabilities{Tools:&ToolsCapability{}}),nil}
 return json.Marshal(ToolsListResult{ResultType:ResultTypeComplete,TTLMS:JSONNumber("0"),CacheScope:CacheScopePrivate,Tools:[]MCPTool{{Name:name,InputSchema:json.RawMessage(`{"type":"object"}`)}}})
 })
 port:=&r13BPublicationPort{bindingContractTransport:&bindingContractTransport{contractTransport:base,bindings:make(map[string][]HTTPToolHeaderBinding)},entered:make(chan struct{}),resume:make(chan struct{})}
 c,err:=Connect(t.Context(),port); require.NoError(t,err); defer c.Close()
 _,err=c.DiscoverTools(t.Context()); require.NoError(t,err)
 name="new"; port.block=true
 ctx,cancel:=context.WithCancel(t.Context()); defer cancel()
 done:=make(chan error,1)
 // Act.
 go func(){_,err:=c.DiscoverTools(ctx);done<-err}()
 <-port.entered; cancel(); close(port.resume)
 err=<-done
 // Assert: cancellation after the documented synchronous commit begins does not roll back.
 require.NoError(t,err)
 port.bindingsMu.Lock();require.Contains(t,port.bindings,"new");require.NotContains(t,port.bindings,"old");port.bindingsMu.Unlock()
}

func TestR13BPageSnapshotDoesNotAuthorizePublicCall(t *testing.T){
 // Arrange.
 calls:=0
 base:=newContractTransport(func(method string,_ json.RawMessage)(json.RawMessage,error){
 switch method {
 case MethodServerDiscover:return completeDiscovery(ServerCapabilities{Tools:&ToolsCapability{}}),nil
 case MethodToolsList:return json.RawMessage(`{"resultType":"complete","ttlMs":123,"cacheScope":"private","tools":[{"name":"one","inputSchema":{"type":"object"},"outputSchema":{"type":"integer"}}]}`),nil
 case MethodToolsCall:calls++;return json.RawMessage(`{"resultType":"complete","content":[],"structuredContent":7}`),nil
 }
 t.Fatalf("unexpected %s",method);return nil,nil
 })
 port:=&bindingContractTransport{contractTransport:base,bindings:make(map[string][]HTTPToolHeaderBinding)}
 c,err:=Connect(t.Context(),port);require.NoError(t,err);defer c.Close()
 // Act/Assert: inspecting even terminal page does not authorize tools/call.
 page,err:=c.ListToolsPage(t.Context(),"");require.NoError(t,err);require.Equal(t,JSONNumber("123"),page.TTLMS)
 _,_,err=c.CallTool(t.Context(),ToolsCallParams{Name:"one",Arguments:json.RawMessage(`{}`)});require.ErrorContains(t,err,"authority");require.Zero(t,calls)
 snapshot,err:=c.DiscoverTools(t.Context());require.NoError(t,err);require.Len(t,snapshot.Pages,1);require.Equal(t,JSONNumber("123"),snapshot.Pages[0].TTLMS)
 raw,err:=json.Marshal(snapshot);require.NoError(t,err);var restored ToolDiscovery;require.NoError(t,json.Unmarshal(raw,&restored));require.Equal(t,snapshot,restored)
 _,_,err=c.CallTool(t.Context(),ToolsCallParams{Name:"one",Arguments:json.RawMessage(`{}`)});require.NoError(t,err);require.Equal(t,1,calls)
}

func TestR13BCancelInProgressStillValidatesMalformedResource(t *testing.T){
 // Arrange: pause the advisory cancellation notification while A is locally canceled
 // but before its await goroutine transfers the route to the retirement map.
 c,tr:=retirementClient(t)
 a,err:=c.Listen(t.Context(),SubscriptionFilter{ToolsListChanged:true});require.NoError(t,err)
 b,err:=c.Listen(t.Context(),SubscriptionFilter{ToolsListChanged:true});require.NoError(t,err);defer b.Close()
 entered,resume:=make(chan struct{}),make(chan struct{})
 tr.mu.Lock();tr.notifyHook=func(n capturedNotification){
 if n.Method!=MethodCancelled{return};var params CancelledParams;_ = json.Unmarshal(n.Params,&params)
 if string(params.RequestID)==a.ID{close(entered);<-resume}
 };tr.mu.Unlock()
 // Act.
 a.Close();<-entered
 require.NoError(t,tr.emit(MethodResourceUpdated,map[string]any{"_meta":Meta{metaSubscriptionID:json.RawMessage(a.ID)}}))
 // Assert: missing mandatory URI must not become valid solely due to cancel timing.
 failed:=b.Err()!=nil
 close(resume);<-a.Done()
 require.True(t,failed,"malformed resource ignored during local-cancel transition")
}

func TestR13BCancellationBeforePublicationCommitPreservesAuthority(t *testing.T){
 // Arrange: hold the same authority lock that publication must acquire.
 name:="old"; fetched:=make(chan struct{},1)
 base:=newContractTransport(func(method string,_ json.RawMessage)(json.RawMessage,error){
 if method==MethodServerDiscover{return completeDiscovery(ServerCapabilities{Tools:&ToolsCapability{}}),nil}
 raw,err:=json.Marshal(ToolsListResult{ResultType:ResultTypeComplete,TTLMS:JSONNumber("0"),CacheScope:CacheScopePrivate,Tools:[]MCPTool{{Name:name,InputSchema:json.RawMessage(`{"type":"object"}`)}}})
 if name=="new" {fetched<-struct{}{}}
 return raw,err
 })
 port:=&bindingContractTransport{contractTransport:base,bindings:make(map[string][]HTTPToolHeaderBinding)}
 c,err:=Connect(t.Context(),port);require.NoError(t,err);defer c.Close()
 _,err=c.DiscoverTools(t.Context());require.NoError(t,err)
 name="new";ctx,cancel:=context.WithCancel(t.Context());defer cancel();done:=make(chan error,1)
 c.toolBindingMu.Lock()
 // Act: canceled while waiting to enter commit; no publication facet invocation.

 err=<-done

func TestR13BPublicTypedCountAndByteRefusals(t *testing.T){
 for _,bytes:=range []bool{false,true}{
 // Arrange.
 limits:=TransportLimits{MaxInFlight:1}
 if bytes{limits=TransportLimits{MaxQueueBytes:1}}
 tr:=NewStdioTransport("/bin/sh",[]string{"-c","sleep 30"},WithStdioLimits(limits));require.NoError(t,tr.Start(t.Context()))
 var first PreparedRequest;var err error
 if !bytes {first,err=tr.PrepareRequest(t.Context(),"test",migrationDiscoveryParams());require.NoError(t,err)}
 // Act.
 _,err=tr.PrepareRequest(t.Context(),"test",migrationDiscoveryParams())
 // Assert.
 require.ErrorIs(t,err,ErrTransportLimitExceeded);var detail *TransportLimitError;require.ErrorAs(t,err,&detail)
 if bytes{require.Equal(t,"queued bytes",detail.Resource);require.Equal(t,1,detail.Limit)}else{require.Equal(t,"in-flight requests",detail.Resource);require.Equal(t,1,detail.Limit);require.NoError(t,first.Abort(context.Canceled))}
 require.NoError(t,tr.Close())
 }
