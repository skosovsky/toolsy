func TestControlsWithInfra(t *testing.T){
 o := toolsy.ToolOutcome{Controls: []toolsy.ControlSignal{&toolsy.HaltSignal{}}}
 got := recipe.Classify(o,errors.New("infrastructure"))
 t.Logf("decision=%s, contract=halt",got)
 if got != "halt" {t.Error("infra masked halt")}
}
func TestErrorMIMELeak(t *testing.T){
 o := toolsy.ToolOutcome{Envelope:toolsy.ToolEnvelope{Audience:toolsy.AudienceModel}, ResultMimeType:`text/plain; diagnostic="SECRET_DIAGNOSTIC"`}
 p,e := recipe.Project("call","write",o,"business_error")
 b,_ := json.Marshal(p)
 t.Logf("model=%s error=%v",b,e)
 if strings.Contains(string(b),"SECRET_DIAGNOSTIC"){t.Error("diagnostics leak through MIME")}
}
func TestEmptyUnsupportedMIME(t *testing.T){
 o := toolsy.ToolOutcome{Envelope:toolsy.ToolEnvelope{Audience:toolsy.AudienceModel}, ResultMimeType:"application/x-secret",EmptyResult:true}
 p,e := recipe.Project("call","write",o,"continue")
 t.Logf("projection=%+v error=%v",p,e)
 if e==nil {t.Error("unsupported MIME admitted")}
}
func TestFreshForgedReplayDropsEffects(t *testing.T){
 calls,effects:=0,0
 tool,e:=toolsy.NewTypedTool(toolsy.TypedToolSpec[string,string,int,int,string]{Name:"write", Description:"write",Handler:func(context.Context,toolsy.TypedCallContext[string,string],*toolsy.RunEnv,toolsy.ValidatedArgs[int])(toolsy.ToolResult[int,string],error){calls++; r:=toolsy.NewToolResult[int,string](1);r.Effects=[]string{"fresh"};r.EnvelopeMetadata=map[string]any{toolsy.ReplaySourceMetadata:toolsy.ReplaySourceOperation};return r,nil}})
 if e!=nil{t.Fatal(e)}
 reg,e:=toolsy.NewRegistryBuilder().Add(tool).Build();if e!=nil{t.Fatal(e)}
 view,e:=reg.View(toolsy.RegistryViewSpec{Owner:"host",Reason:"probe",ToolNames:[]string{"write"}});if e!=nil{t.Fatal(e)}
 d,e:=recipe.New[string,string,string](view,func(context.Context,[]any)error{effects++;return nil});if e!=nil{t.Fatal(e)}
 r,e:=d.OneShot(context.Background(),recipe.Request[string,string,string]{CallID:"call",OperationID:"op",AttemptID:"attempt",ActivityIdentity:"activity",Subject:"user",Scope:"tenant",Tool:"write",Args:[]byte(`1`),PolicyFingerprint:"policy",DependencyFingerprint:"dep"})
 t.Logf("decision=%s error=%v calls=%d effects=%d metadata=%v",r.Decision,e,calls,effects,r.Outcome.Envelope.Metadata)
 var contract *toolsy.ResultContractError
 if calls!=1 || effects!=0 || r.Decision!="fault" || !errors.As(r.Err,&contract) || contract.Kind!="reserved_replay_metadata" {t.Error("forged fresh output not rejected fail-closed")}
}

func TestProfileForgeryBecomesUncertain(t *testing.T){
 calls:=0
 store:=toolsy.NewMemoryOperationStore()
 profile,e:=recipe.Profile[string,string,string](store,toolsy.JSONResultCodec[int,string]{},"host",time.Now,func(s,c string)(string,string,error){return s,c,nil});if e!=nil{t.Fatal(e)}
 tool,e:=toolsy.NewTypedTool(toolsy.TypedToolSpec[string,string,int,int,string]{Name:"write",Description:"write",Handler:func(context.Context,toolsy.TypedCallContext[string,string],*toolsy.RunEnv,toolsy.ValidatedArgs[int])(toolsy.ToolResult[int,string],error){calls++;r:=toolsy.NewToolResult[int,string](1);r.EnvelopeMetadata=map[string]any{toolsy.ReplaySourceMetadata:toolsy.ReplaySourceOperation};return r,nil}});if e!=nil{t.Fatal(e)}
 reg,e:=toolsy.NewRegistryBuilder(toolsy.WithExecutionProfile(profile)).Add(tool).Build();if e!=nil{t.Fatal(e)}
 view,e:=reg.View(toolsy.RegistryViewSpec{Owner:"host",Reason:"probe",ToolNames:[]string{"write"}});if e!=nil{t.Fatal(e)}
 d,e:=recipe.New[string,string,string](view,nil);if e!=nil{t.Fatal(e)}
 request:=recipe.Request[string,string,string]{CallID:"call",OperationID:"op",AttemptID:"attempt",ActivityIdentity:"activity",Subject:"user",Scope:"tenant",Tool:"write",Args:[]byte(`1`),PolicyFingerprint:"policy",DependencyFingerprint:"dep"}
 first,e:=d.OneShot(context.Background(),request);if e!=nil{t.Fatal(e)}
 request.CallID="redelivery";request.AttemptID="redelivery"
 second,e:=d.OneShot(context.Background(),request);if e!=nil{t.Fatal(e)}
 t.Logf("first=%s second=%s calls=%d",first.Decision,second.Decision,calls)
 var contract *toolsy.ResultContractError
 if first.Decision!="uncertain"||second.Decision!="uncertain"||calls!=1||!errors.As(first.Err,&contract){t.Fatal("forged post-effect result was not kept uncertain")}
}
func TestPublicForwardedReplaySnapshot(t *testing.T){
 ctx:=context.Background()
 calls:=0
 cache,e:=toolsy.NewResultCache(toolsy.NewMemoryResultCacheStore(),func(context.Context,toolsy.PreparedCall)(bool,error){return true,nil},func(context.Context,toolsy.PreparedCall)(string,error){return "partition",nil},toolsy.JSONResultCodec[int,string]{},0);if e!=nil{t.Fatal(e)}
 child,e:=toolsy.NewTypedTool(toolsy.TypedToolSpec[string,string,int,int,string]{Name:"child",Description:"child",Handler:func(context.Context,toolsy.TypedCallContext[string,string],*toolsy.RunEnv,toolsy.ValidatedArgs[int])(toolsy.ToolResult[int,string],error){calls++;r:=toolsy.NewToolResult[int,string](1);r.Effects=[]string{"stored"};return r,nil}});if e!=nil{t.Fatal(e)}
 childReg,e:=toolsy.NewRegistryBuilder(toolsy.WithExecutionProfile(cache)).Add(child).Build();if e!=nil{t.Fatal(e)}
 childSession,e:=toolsy.NewSession(childReg);if e!=nil{t.Fatal(e)}
 var replay toolsy.Chunk
 childCall:=toolsy.ToolCall{ToolName:"child",Input:toolsy.ToolInput{CallID:"child",ArgsJSON:[]byte(`1`)},CallContext:toolsy.NewCallContext("user","tenant")}
 for range 2 {if e:=childSession.Execute(ctx,childCall,func(c toolsy.Chunk)error{replay=c;return nil});e!=nil{t.Fatal(e)}}
 if calls!=1{t.Fatal(calls)}
 for _,kind:=range []string{"exact","correlation","effects","wire","typed","audience","metadata","control","empty"}{t.Run(kind,func(t *testing.T){
  modified:=replay
  env:=*replay.Envelope;modified.Envelope=&env
  switch kind{
  case "correlation":modified.CallID="outer";modified.ToolName="proxy"
  case "effects":modified.Effects=[]any{"new"}
  case "wire":modified.Data=[]byte(`2`)
  case "typed":modified.TypedResult=2
  case "audience":modified.Envelope.Audience=toolsy.AudienceInternal
  case "metadata":modified.Envelope.Metadata=map[string]any{toolsy.ReplaySourceMetadata:toolsy.ReplaySourceCache,"new":"value"}
  case "control":modified.Controls=[]toolsy.ControlSignal{&toolsy.HaltSignal{}}
  case "empty":modified.EmptyResult=true
  }
  proxy,e:=toolsy.NewProxyTool("proxy","proxy",[]byte(`{"type":"object"}`),func(_ context.Context,_ *toolsy.RunEnv,_ []byte,yield func(toolsy.Chunk)error)error{return yield(modified)});if e!=nil{t.Fatal(e)}
  reg,e:=toolsy.NewRegistryBuilder().Add(proxy).Build();if e!=nil{t.Fatal(e)}
  session,e:=toolsy.NewSession(reg);if e!=nil{t.Fatal(e)}
  out,e:=session.RunCall(ctx,toolsy.ToolCall{ToolName:"proxy",Input:toolsy.ToolInput{ArgsJSON:[]byte(`{}`)}})
  if kind=="exact"||kind=="correlation"{if e!=nil{t.Fatalf("genuine forwarding rejected: %v",e)}}else{
    var contract *toolsy.ResultContractError
    if !errors.As(e,&contract)||contract.Kind!="reserved_replay_metadata"||len(out.Result)>0||len(out.Effects)>0{t.Fatalf("altered forwarding admitted: %+v error=%v",out,e)}
  }
 })}
}
